\begin{frame}{The idea in 30 seconds}
  \begin{center}
    \small
    PROBLEM $\rightarrow$ hypotheses $\rightarrow$ methods $\rightarrow$ kanban\\
    $\rightarrow$ report $\rightarrow$ verdict + insights $\rightarrow$ knowledge base
  \end{center}
  \vspace{0.5em}
  \begin{block}{Core principle}
    Knowledge comes only from an experiment with a \textbf{predefined decision rule}.
    A verdict substitutes measured values into the rule; it is not the model's impression.
  \end{block}
  \begin{itemize}
    \item Storage: Markdown files, no database
    \item UI: local editor for \texttt{project.md}, reports, and kanban
    \item The IDE agent is the default executor
  \end{itemize}
\end{frame}

\begin{frame}{Insight 1: portable skills}
  \begin{block}{Skills can move between Claude, Cursor, and Codex}
    \texttt{SKILL.md} is a universal playbook format;\\
    \texttt{AGENTS.md} is a shared entry point for any agentic IDE.
  \end{block}
  \begin{itemize}
    \item Shared playbooks live in \texttt{agents/skills/} and are not tied to Cursor
    \item The agent adapts paths and commands to its environment
    \item One repository provides the same workflows to the entire team
  \end{itemize}
\end{frame}

\begin{frame}{Insight 2: decentralized synchronization}
  \begin{block}{Git plus an agent instead of a shared host}
    Everyone has a local instance at localhost:8080;\\
    agents initiate synchronization of project data through Git.
  \end{block}
  \begin{itemize}
    \item Only \texttt{projects/<id>/} is synchronized
    \item \texttt{koi-project-sync}: pull at startup and push meaningful changes
    \item \texttt{sessionStart} and \texttt{stop} hooks require no separate CI
  \end{itemize}
\end{frame}

\begin{frame}{Insight 3: a web platform without an agent API}
  \begin{block}{Hooks, Inbox, and a file queue}
    UI $\rightarrow$ \texttt{agent-chat-queue.json} $\rightarrow$ watcher $\rightarrow$ agent $\rightarrow$ UI answer
  \end{block}
  \begin{tabular}{lccc}
    \toprule
    Mode & Latency & API & One chat \\
    \midrule
    Inbox chat & $\sim$1--3\,s & no & yes \\
    Hooks & up to minutes & no & no \\
    Background API & $\sim$5\,s & yes & no \\
    \bottomrule
  \end{tabular}
  \vspace{0.5em}
  \small Inbox: JSON queue $\rightarrow$ \texttt{koi\_agent\_chat\_inbox.py pending} $\rightarrow$ \texttt{koi-agent-chat} or \texttt{koi-related-work}
\end{frame}

\begin{frame}{Insight 4: separate intelligence from transport}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \textbf{The agent performs:}
    \begin{itemize}
      \item experiments
      \item analysis and synthesis
      \item knowledge curation
      \item answers in the UI chat
    \end{itemize}
    \column{0.48\textwidth}
    \textbf{The code performs:}
    \begin{itemize}
      \item report-format validation
      \item idempotent ingestion
      \item rebuilding \texttt{KNOWLEDGE.md}
      \item queues and hooks
    \end{itemize}
  \end{columns}
  \vspace{0.5em}
  \begin{alertblock}{Format gate}
    An invalid \texttt{.run.md} is rejected in full; state remains unchanged.
  \end{alertblock}
\end{frame}

\begin{frame}{Insight 5: a two-layer knowledge base}
  \begin{block}{Generated layer (rules)}
    \texttt{KNOWLEDGE.md}, \texttt{hypotheses.md}, \texttt{KNOWLEDGE\_LOG.md}\\
    Exact, flat, and rebuilt on every save
  \end{block}
  \begin{block}{Curated layer (agent)}
    \texttt{projects/<id>/knowledge/*.md}\\
    Cross-analysis: connections, contradictions, applicability limits, and open questions
  \end{block}
  \small Chat reads \texttt{research.json} first and opens reports only for missing detail.
\end{frame}

\begin{frame}{Insight 6: file queues as an orchestrator}
  \begin{itemize}
    \item \texttt{agent-chat-queue.json}: questions from the UI
    \item done-research queue: cards awaiting a finding
    \item sync-push queue: changes to commit to Git
  \end{itemize}
  \vspace{0.5em}
  At session start, the agent:
  \begin{enumerate}
    \item pulls synchronized state
    \item checks every queue
    \item optionally starts the development server
  \end{enumerate}
  \vspace{0.5em}
  \textbf{No central scheduler}: only files, hooks, and agent discipline.
\end{frame}

\begin{frame}{Insight 7: the agent as UI QA}
  \begin{itemize}
    \item \texttt{koi-ui-design-review}: lint, design tokens, and accessibility
    \item \texttt{koi-visual-qa}: screenshots and mind-map/kanban/modal regression
    \item The agent tests the same UI it writes
  \end{itemize}
\end{frame}

\begin{frame}{Insight 8: graceful degradation}
  \begin{block}{A clone without a cloud backend is normal}
    If \texttt{koi\_check\_hypothesis.py} is unavailable, the agent itself:
  \end{block}
  \begin{enumerate}
    \item runs the experiment locally
    \item fills in \texttt{experiment-report.md}
    \item runs \texttt{--ingest-only}, starting with \texttt{--dry-run}
  \end{enumerate}
  The repository is designed so the \textbf{IDE agent can replace the server worker}.
\end{frame}

\begin{frame}{Conclusions}
  \begin{enumerate}
    \item \textbf{Markdown + Git + queues} form a minimal agent-integration stack
    \item \textbf{Skills} are portable playbooks, not an IDE lock-in
    \item \textbf{Format gates} keep the LLM from writing directly to the knowledge base
    \item \textbf{Agent = researcher + synchronizer + QA}; code = transport
    \item \textbf{Local host + Git} enables teamwork without a shared server
  \end{enumerate}
\end{frame}
